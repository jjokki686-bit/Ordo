name: iOS App Build

on:
  workflow_dispatch: # 버튼 클릭으로 수동 빌드 허용
  push:
    branches: [ "main" ]

jobs:
  build:
    runs-on: macos-14 # GitHub가 제공하는 가상 Mac

    steps:
      - name: 1. 저장소 소스코드 가져오기
        uses: actions/checkout@v4

      - name: 2. Node.js 환경 세팅
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: 3. 의존성 설치 및 동기화
        run: |
          npm install
          npx cap sync ios

      - name: 4. Xcode로 iOS 앱 빌드 검증
        run: |
          cd ios/App
          xcodebuild -workspace App.xcworkspace \
                     -scheme App \
                     -destination generic/platform=iOS \
                     -configuration Release \
                     CODE_SIGNING_ALLOWED=NO \
                     build
